import json,os,subprocess,tempfile,uuid
from pathlib import Path
from factory.harness import load_harness_config
names=set(load_harness_config(Path('/Users/james/factory')).secret_vars)
root=Path(tempfile.mkdtemp(prefix='factory-learning-kit-'));kit=root/'kit';kit.mkdir();workspace=root/'workspace';workspace.mkdir()
name='factory-build-learning-kit-'+uuid.uuid4().hex[:8]
# The kit overrides declared sensitive names with empty strings; no secret values read.
spec='schemaVersion: "1"\nkind: mixin\nname: learning-empty-env\ndescription: Empty declared sensitive environment for isolated learning verification\nenvironment:\n  variables:\n'+''.join('    '+n+': ""\n' for n in sorted(names))
(kit/'spec.yaml').write_text(spec)
r=subprocess.run(['sbx','kit','validate',str(kit)],capture_output=True,text=True);print('validate',r.returncode,r.stdout,r.stderr,flush=True)
if r.returncode:raise SystemExit(r.returncode)
env={k:v for k,v in os.environ.items() if k not in names}
argv=['sbx','create','--name',name,'--kit',str(kit),'--deny-network','api.github.com','--deny-network','github.com','codex',str(workspace)]
print('root',root,'sandbox',name,flush=True)
r=subprocess.run(argv,env=env,capture_output=True,text=True,timeout=180)
print('create',r.returncode,flush=True)
results={'name':name,'root':str(root),'create_exit':r.returncode}
if not r.returncode:
 sock='unix:///Users/james/Library/Application Support/com.docker.sandboxes/sandboxes/sandboxd/docker.sock'
 r=subprocess.run(['docker','-H',sock,'container','inspect',name],capture_output=True,text=True,check=True)
 d=json.loads(r.stdout)[0]
 results['container_nonempty']=[e.partition('=')[0] for e in d.get('Config',{}).get('Env',[]) if e.partition('=')[0] in names and e.partition('=')[2]]
 r=subprocess.run(['sbx','inspect',name,'--json'],capture_output=True,text=True,check=True)
 d=json.loads(r.stdout);results['proxy_secrets']=d.get('secrets',[])
 code='import os,json;names='+repr(sorted(names))+';print(json.dumps([n for n in names if os.environ.get(n)]))'
 r=subprocess.run(['sbx','exec',name,'python3','-c',code],capture_output=True,text=True,timeout=60)
 results['direct_exec_exit']=r.returncode;results['direct_exec_nonempty']=json.loads(r.stdout) if r.returncode==0 else None
 if results['container_nonempty'] or results['direct_exec_nonempty']:
  results['remove_exit']=subprocess.run(['sbx','rm','--force',name],capture_output=True).returncode
else:results['create_error']=r.stderr[-1200:]
(root/'result.json').write_text(json.dumps(results,indent=2));print(json.dumps(results,indent=2))
